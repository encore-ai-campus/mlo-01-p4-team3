```
# (참고) Windows 로컬 Spark 세션: JAVA_HOME=JDK 17
venv314\Scripts\python -c "from pyspark.sql import SparkSession; s=SparkSession.builder.master('local[2]').getOrCreate(); print('spark', s.version, s.range(5).count(), s.createDataFrame([(1,)],'a int').selectExpr('a+1 as b').collect())"
# (조사 항목) 휠 존재만: venv314\Scripts\python -m pip install --dry-run --only-binary=:all: <패키지>
venv314\Scripts\python -m pip install --dry-run --only-binary=:all: -r requirements/web.txt -r requirements/ml.txt -r requirements/llm.txt -r requirements/pipeline.txt

# 조건 3: Spark 4.1.3 + Delta 4.4.1
docker run --rm --memory 1g -v <저장소>/ops/compose/spark:/work:ro apache/spark:4.1.3-scala2.13-java17-python3-ubuntu \
  /opt/spark/bin/spark-submit --master local[2] --driver-memory 512m --conf spark.jars.ivy=/tmp/.ivy2 \
  --packages io.delta:delta-spark_2.13:4.4.1 \
  --conf spark.sql.extensions=io.delta.sql.DeltaSparkSessionExtension \
  --conf spark.sql.catalog.spark_catalog=org.apache.spark.sql.delta.catalog.DeltaCatalog /work/delta_rw.py /tmp/delta_smoke
# 위와 같고 --packages io.delta:delta-spark_4.1_2.13:4.4.1 (대안: 같은 Delta 버전의 Spark 4.1용 jar)

# 조건 4: .env 추적 여부, 비밀값 노출 검사
git check-ignore -v .env
python -m unittest tests.ops.test_devenv     # .env.example 값이 자리표시자뿐인지, dev.yml에 비밀값이 없는지
(.env의 12자 이상 값을 Git 추적·미추적 파일 120개에서 검색하는 스크립트)

# 전체 테스트
python -m unittest discover -s tests -t .
```
